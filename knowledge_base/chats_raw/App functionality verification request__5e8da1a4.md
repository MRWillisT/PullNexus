def update_chokepoint_data(self, chokepoint_status: dict) -> None:
    """
    Called by main window when AIS data updates.
    chokepoint_status keys are full names like 'Strait of Hormuz'.
    Map them to flow node names.
    """
    NAME_MAP = {
        "Strait of Hormuz":   "HORMUZ",
        "Suez Canal":         "SUEZ",
        "Strait of Malacca":  "MALACCA",
        "Bab-el-Mandeb":      "BAB-EL-MANDEB",
        "Bosphorus":          "BOSPHORUS",
        "Panama Canal":       "PANAMA",
        "Cape of Good Hope":  None,  # not a flow node
    }
    for full_name, node_name in NAME_MAP.items():
        if node_name and full_name in chokepoint_status:
            data = chokepoint_status[full_name]
            self._chokepoint_counts[node_name] = data.get("count", 0)
            self._chokepoint_anomaly[node_name] = data.get("anomaly", False)
    self.update()

def _draw_vessel_badge(
    self,
    painter: QPainter,
    node_pt: QPointF,
    count: int,
    anomaly: bool,
) -> None:
    """
    Draw a small vessel count badge above the node circle.
    Shows ship emoji + count. Red background if anomaly detected.
    """
    painter.save()

    badge_text = f"🚢 {count}"
    bg_color   = QColor("#cc2200" if anomaly else "#0d2e1e")
    border_col = QColor("#ff4422" if anomaly else "#39ff88")
    text_col   = QColor("#ffffff" if anomaly else "#b9ffd8")

    font = QFont("Segoe UI", 7)
    font.setBold(True)
    painter.setFont(font)

    fm     = painter.fontMetrics()
    text_w = fm.horizontalAdvance(badge_text)
    text_h = fm.height()

    pad_x, pad_y = 5, 2
    badge_w = text_w + pad_x * 2
    badge_h = text_h + pad_y * 2

    bx = int(node_pt.x() - badge_w / 2)
    by = int(node_pt.y() - 18 - badge_h)

    # Background
    painter.setPen(Qt.PenStyle.NoPen)
    painter.setBrush(bg_color)
    painter.drawRoundedRect(bx, by, badge_w, badge_h, 3, 3)

    # Border
    painter.setPen(QPen(border_col, 1))
    painter.setBrush(Qt.BrushStyle.NoBrush)
    painter.drawRoundedRect(bx, by, badge_w, badge_h, 3, 3)

    # Text
    painter.setPen(text_col)
    painter.drawText(bx + pad_x, by + pad_y + fm.ascent(), badge_text)

    # Pulse ring if anomaly
    if anomaly:
        pulse_color = QColor("#ff4422")
        pulse_color.setAlpha(60 + (self._flash_tick % 20) * 8)
        painter.setPen(QPen(pulse_color, 2))
        painter.setBrush(Qt.BrushStyle.NoBrush)
        r = 14 + (self._flash_tick % 10)
        painter.drawEllipse(node_pt, r, r)

    painter.restore()
